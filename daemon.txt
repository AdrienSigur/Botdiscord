/etc/systemd/system/bot.service

[Unit]
Description=Discord Bot Python
After=network.target

[Service]
User=$USER
Group=$USER
WorkingDirectory=/home/$USER/botdiscord
ExecStart=/home/$USER/botdiscord/env/bin/python /home/$USER/botdiscord/bot.py
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target[Unit]
